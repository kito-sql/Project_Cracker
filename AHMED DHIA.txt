#include <iostream>
#include <vector>
using namespace std;

int main() {

    // Corrections : {wrong, right, misplaced}
    int corr[5][3] = {
        {1,1,1},
        {3,0,0},
        {2,1,0},
        {1,0,2},
        {1,1,1}
    };

    // Tentatives
    int att[5][3] = {
        {2,3,4},
        {5,6,7},
        {5,6,4},
        {4,0,1},
        {1,2,4}
    };

    // 1) Récupération des chiffres possibles
    vector<int> digits;
    bool exist[10] = {false};

    for (int i = 0; i < 5; i++) {
        for (int j = 0; j < 3; j++) {
            int d = att[i][j];
            if (!exist[d]) {
                exist[d] = true;
                digits.push_back(d);
            }
        }
    }



    int solutions = 0;

    // 2) Tester seulement les combinaisons possibles
    for (int x : digits) {
        for (int y : digits) {
            for (int z : digits) {

                int code[3] = {x, y, z};
                bool ok = true;

                // Vérification avec les 5 tentatives
                for (int t = 0; t < 5; t++) {

                    int right = 0, misplaced = 0;
                    bool usedC[3] = {0}, usedG[3] = {0};

                    // Bons chiffres bien placés
                    for (int i = 0; i < 3; i++) {
                        if (code[i] == att[t][i]) {
                            right++;
                            usedC[i] = usedG[i] = true;
                        }
                    }

                    // Bons chiffres mal placés
                    for (int i = 0; i < 3; i++) {
                        if (!usedG[i]) {
                            for (int j = 0; j < 3; j++) {
                                if (!usedC[j] && att[t][i] == code[j]) {
                                    misplaced++;
                                    usedC[j] = usedG[i] = true;
                                    break;
                                }
                            }
                        }
                    }

                    int wrong = 3 - right - misplaced;

                    if (wrong != corr[t][0] ||
                        right != corr[t][1] ||
                        misplaced != corr[t][2]) {
                        ok = false;
                        break;
                    }
                }

                if (ok) {
                    cout << "Code valide : " << x << " " << y << " " << z << endl;
                    solutions++;
                }
            }
        }
    }



    return 0;
}
